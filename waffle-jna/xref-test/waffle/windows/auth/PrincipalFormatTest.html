18       * Test known.

20      @Test
21      void testKnown() {
22          Assertions.assertEquals(PrincipalFormat.FQN, PrincipalFormat.valueOf("FQN"));
23          Assertions.assertEquals(PrincipalFormat.SID, PrincipalFormat.valueOf("SID"));
24          Assertions.assertEquals(PrincipalFormat.BOTH, PrincipalFormat.valueOf("BOTH"));
25          Assertions.assertEquals(PrincipalFormat.NONE, PrincipalFormat.valueOf("NONE"));
26          Assertions.assertEquals(4, PrincipalFormat.values().length);
27      }
28  
29      /**
