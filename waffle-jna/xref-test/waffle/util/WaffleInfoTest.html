37      /**
38       * Test waffle info.
39       *
40       * @throws ParserConfigurationException
41       *             the parser configuration exception
42       */
43      @Test
44      void testWaffleInfo() throws ParserConfigurationException {
45          final WaffleInfo helper = new WaffleInfo();
46          final Document info = helper.getWaffleInfo();
47  
48          // Make sure JNA Version is properly noted
49          Assertions.assertEquals(Platform.class.getPackage().getImplementationVersion(),
50                  info.getDocumentElement().getAttribute("jna"));
51  
52          // waffle auth currentUser computer
53          final Node node = info.getDocumentElement().getFirstChild().getFirstChild().getNextSibling();
54  
55          Assertions.assertEquals("computer", node.getNodeName());
56  
57          final IWindowsAuthProvider auth = new WindowsAuthProviderImpl();
58          final IWindowsComputer computer = auth.getCurrentComputer();
59  
60          final NodeList nodes = node.getChildNodes();
61          Assertions.assertEquals(computer.getComputerName(), nodes.item(0).getTextContent());
62          Assertions.assertEquals(computer.getMemberOf(), nodes.item(1).getTextContent());
63          Assertions.assertEquals(computer.getJoinStatus(), nodes.item(2).getTextContent());
64  
65          // Add Lookup Info for Various accounts
66          String lookup = WindowsAccountImpl.getCurrentUsername();
67          final IWindowsAccount account = new WindowsAccountImpl(lookup);
68          Element elem = helper.getLookupInfo(info, lookup);
69          Assertions.assertEquals(lookup, elem.getAttribute("name"));
70          Assertions.assertEquals(account.getName(), elem.getFirstChild().getTextContent());
71  
72          // Report an error when unknown name
73          lookup = "__UNKNOWN_ACCOUNT_NAME___";
74          elem = helper.getLookupInfo(info, lookup);
75          Assertions.assertEquals(lookup, elem.getAttribute("name"));
76          Assertions.assertEquals("exception", elem.getFirstChild().getNodeName());
77      }

79      /**
80       * Test get exception element.
81       *
82       * @throws ParserConfigurationException
83       *             the parser configuration exception
84       */
85      @Test
86      void testGetException() throws ParserConfigurationException {
87          final Document doc = DocumentBuilderFactory.newInstance().newDocumentBuilder().newDocument();
88          final RuntimeException ex = new RuntimeException("test error message");
89          final Element exElem = WaffleInfo.getException(doc, ex);
90  
91          Assertions.assertNotNull(exElem);
92          Assertions.assertEquals("exception", exElem.getNodeName());
93          Assertions.assertEquals(ex.getClass().getName(), exElem.getAttribute("class"));
94  
95          // message child
96          final Node messageNode = exElem.getFirstChild();
97          Assertions.assertEquals("message", messageNode.getNodeName());
98          Assertions.assertEquals("test error message", messageNode.getTextContent());
99      }

136      * Test to pretty xml with a manually created document (no Windows dependency).
137      *
138      * @throws ParserConfigurationException
139      *             the parser configuration exception
140      * @throws TransformerException
141      *             the transformer exception
142      */
143     @Test
144     void testToPrettyXmlWithManualDocument() throws ParserConfigurationException, TransformerException {
145         final DocumentBuilderFactory df = DocumentBuilderFactory.newInstance();
146         df.setAttribute(XMLConstants.ACCESS_EXTERNAL_DTD, "");
147         df.setAttribute(XMLConstants.ACCESS_EXTERNAL_SCHEMA, "");
148         final Document doc = df.newDocumentBuilder().newDocument();
149         final Element root = doc.createElement("waffle");
150         doc.appendChild(root);
151         root.setAttribute("test", "true");
152 
153         final String xml = WaffleInfo.toPrettyXML(doc);
154 
155         Assertions.assertNotNull(xml);
156         Assertions.assertFalse(xml.isEmpty());
157         Assertions.assertTrue(xml.contains("waffle"));
158         Assertions.assertTrue(xml.contains("test"));
159     }

161     /**
162      * Test add account info populates all fields.
163      *
164      * @param account
165      *            the account
166      *
167      * @throws ParserConfigurationException
168      *             the parser configuration exception
169      */
170     @Test
171     void testAddAccountInfo(@Mocked final IWindowsAccount account) throws ParserConfigurationException {
172         final DocumentBuilderFactory df = DocumentBuilderFactory.newInstance();
173         df.setAttribute(XMLConstants.ACCESS_EXTERNAL_DTD, "");
174         df.setAttribute(XMLConstants.ACCESS_EXTERNAL_SCHEMA, "");
175         final Document doc = df.newDocumentBuilder().newDocument();
176         final Element node = doc.createElement("testNode");
177 
178         Assertions.assertNotNull(new Expectations() {
179             {
180                 account.getName();
181                 this.result = "testuser";
182                 account.getFqn();
183                 this.result = "DOMAIN\\testuser";
184                 account.getDomain();
185                 this.result = "DOMAIN";
186                 account.getSidString();
187                 this.result = "S-1-5-21-123";
188             }
189         });
190 
191         final WaffleInfo helper = new WaffleInfo();
192         helper.addAccountInfo(doc, node, account);
193 
194         final NodeList children = node.getChildNodes();
195         Assertions.assertEquals(4, children.getLength());
196         Assertions.assertEquals("Name", children.item(0).getNodeName());
197         Assertions.assertEquals("testuser", children.item(0).getTextContent());
198         Assertions.assertEquals("FQN", children.item(1).getNodeName());
199         Assertions.assertEquals("DOMAIN\\testuser", children.item(1).getTextContent());
200         Assertions.assertEquals("Domain", children.item(2).getNodeName());
201         Assertions.assertEquals("DOMAIN", children.item(2).getTextContent());
202         Assertions.assertEquals("SID", children.item(3).getNodeName());
203         Assertions.assertEquals("S-1-5-21-123", children.item(3).getTextContent());
204     }
