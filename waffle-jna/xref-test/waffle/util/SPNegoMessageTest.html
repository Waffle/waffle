7   package waffle.util;
8   
9   import org.junit.jupiter.api.Assertions;
10  import org.junit.jupiter.api.Test;
11  
12  /**
13   * The Class SPNegoMessageTest.
14   */
15  class SPNegoMessageTest {
16  
17      // Different SPNEGO messages. For details and specification,
18      // see https://msdn.microsoft.com/en-us/library/ms995330.aspx
19  
20      /** The Constant NEG_TOKEN_INIT_OK. */
21      private static final byte[] NEG_TOKEN_INIT_OK = { 0x60, 0x76, 0x06, 0x06, 0x2B, 0x06, 0x01, 0x05, 0x05, 0x02 };
22  
23      /** The Constant NEG_TOKEN_INIT_TOO_SHORT. */
24      private static final byte[] NEG_TOKEN_INIT_TOO_SHORT = { 0x60, 0x76, 0x06, 0x06, 0x2B, 0x06, 0x01, 0x05, 0x05 };
25  
26      /** The Constant NEG_TOKEN_ARG_OK. */
27      private static final byte[] NEG_TOKEN_ARG_OK = { (byte) 0xA1, 0x33, 0x30, 0x31, 0x0, 0x03, 0x0A, 0x01, 0x01, 0x0,
28              0x2A, 0x04, 0x28, 0x4E, 0x54, 0x4C, 0x4D, 0x53, 0x53, 0x50, 0x00, 0x01, 0x00, 0x00, 0x00, 0x0, 0x0, 0x08,
29              0x0, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x06,
30              0x01, 0x0, 0x1D, 0x00, 0x00, 0x00, 0x0F };
31  
32      /** The Constant NEG_TOKEN_ARG_TOO_SHORT. */
33      private static final byte[] NEG_TOKEN_ARG_TOO_SHORT = { (byte) 0xA1, 0x33, 0x30, 0x31, 0x0, 0x03 };
34  
35      /** The Constant BAD_MESSAGE. */
36      private static final byte[] BAD_MESSAGE = { 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00 };
37  
38      /**
39       * Test is neg token init.
40       */
41      @Test
42      void testIsNegTokenInit() {
43          Assertions.assertTrue(SPNegoMessage.isNegTokenInit(SPNegoMessageTest.NEG_TOKEN_INIT_OK));
44          Assertions.assertFalse(SPNegoMessage.isNegTokenInit(SPNegoMessageTest.NEG_TOKEN_INIT_TOO_SHORT));
45          Assertions.assertFalse(SPNegoMessage.isNegTokenInit(SPNegoMessageTest.BAD_MESSAGE));
46      }
47  
48      /**
49       * Test is neg token arg.
50       */
51      @Test
52      void testIsNegTokenArg() {
53          Assertions.assertTrue(SPNegoMessage.isNegTokenArg(SPNegoMessageTest.NEG_TOKEN_ARG_OK));
54          Assertions.assertFalse(SPNegoMessage.isNegTokenArg(SPNegoMessageTest.NEG_TOKEN_ARG_TOO_SHORT));
55          Assertions.assertFalse(SPNegoMessage.isNegTokenArg(SPNegoMessageTest.BAD_MESSAGE));
56      }
57  
58      /**
59       * Test is neg token init with null returns false.
60       */
61      @Test
62      void testIsNegTokenInitNullReturnsFalse() {
63          Assertions.assertFalse(SPNegoMessage.isNegTokenInit(null));
64      }
65  
66      /**
67       * Test is neg token arg with null returns false.
68       */
69      @Test
70      void testIsNegTokenArgNullReturnsFalse() {
71          Assertions.assertFalse(SPNegoMessage.isNegTokenArg(null));
72      }
73  
74      /**
75       * Test is neg token init with single byte returns false.
76       */
77      @Test
78      void testIsNegTokenInitSingleByteReturnsFalse() {
79          Assertions.assertFalse(SPNegoMessage.isNegTokenInit(new byte[] { 0x60 }));
80      }
81  
82      /**
83       * Test is neg token arg with single byte returns false.
84       */
85      @Test
86      void testIsNegTokenArgSingleByteReturnsFalse() {
87          Assertions.assertFalse(SPNegoMessage.isNegTokenArg(new byte[] { (byte) 0xA1 }));
88      }
89  
90      /**
91       * Test is neg token init with long length encoding.
92       */
93      @Test
94      void testIsNegTokenInitWithLongLengthEncoding() {
95          // Message with multi-byte length (0x81 means 1 length byte follows), then OID
96          final byte[] msg = { 0x60, (byte) 0x81, 0x0A, 0x06, 0x06, 0x2B, 0x06, 0x01, 0x05, 0x05, 0x02, 0x00 };
97          Assertions.assertTrue(SPNegoMessage.isNegTokenInit(msg));
98      }
99  }
