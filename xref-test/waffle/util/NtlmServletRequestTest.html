7   package waffle.util;
8   
9   import org.junit.jupiter.api.Assertions;
10  import org.junit.jupiter.api.Test;
11  
12  import waffle.mock.http.SimpleHttpRequest;
13  
14  /**
15   * The Class NtlmServletRequestTest.
16   */
17  class NtlmServletRequestTest {
18  
19      /**
20       * Test get connection id.
21       */
22      @Test
23      void testGetConnectionId() {
24          SimpleHttpRequest.resetRemotePort();
25          final SimpleHttpRequest request1 = new SimpleHttpRequest();
26          Assertions.assertEquals(":1", NtlmServletRequest.getConnectionId(request1));
27          final SimpleHttpRequest request2 = new SimpleHttpRequest();
28          Assertions.assertEquals(":2", NtlmServletRequest.getConnectionId(request2));
29          request2.setRemoteAddr("192.168.1.1");
30          Assertions.assertEquals("192.168.1.1:2", NtlmServletRequest.getConnectionId(request2));
31          request2.setRemoteHost("codeplex.com");
32          Assertions.assertEquals("codeplex.com:2", NtlmServletRequest.getConnectionId(request2));
33      }
34  }
