7   package waffle.shiro.negotiate;
8   
9   import java.security.Principal;
10  
11  import javax.security.auth.Subject;
12  
13  import org.junit.jupiter.api.Assertions;
14  import org.junit.jupiter.api.BeforeEach;
15  import org.junit.jupiter.api.Test;
16  
17  /**
18   * Tests for {@link NegotiateToken}.
19   */
20  class NegotiateTokenTest {
21  
22      /** The token. */
23      private NegotiateToken token;
24  
25      /** The in bytes. */
26      private static final byte[] IN_BYTES = { 0x01, 0x02, 0x03 };
27  
28      /** The out bytes. */
29      private static final byte[] OUT_BYTES = { 0x04, 0x05, 0x06 };
30  
31      /**
32       * Sets the up.
33       */
34      @BeforeEach
35      void setUp() {
36          this.token = new NegotiateToken(NegotiateTokenTest.IN_BYTES, NegotiateTokenTest.OUT_BYTES, "conn-001",
37                  "Negotiate", false, false, "localhost");
38      }
39  
40      /**
41       * Test get connection id.
42       */
43      @Test
44      void testGetConnectionId() {
45          Assertions.assertEquals("conn-001", this.token.getConnectionId());
46      }
47  
48      /**
49       * Test get security package.
50       */
51      @Test
52      void testGetSecurityPackage() {
53          Assertions.assertEquals("Negotiate", this.token.getSecurityPackage());
54      }
55  
56      /**
57       * Test is ntlm post false.
58       */
59      @Test
60      void testIsNtlmPostFalse() {
61          Assertions.assertFalse(this.token.isNtlmPost());
62      }
63  
64      /**
65       * Test is ntlm post true.
66       */
67      @Test
68      void testIsNtlmPostTrue() {
69          final NegotiateToken ntlmToken = new NegotiateToken(NegotiateTokenTest.IN_BYTES, NegotiateTokenTest.OUT_BYTES,
70                  "conn-001", "NTLM", true, false, null);
71          Assertions.assertTrue(ntlmToken.isNtlmPost());
72      }
73  
74      /**
75       * Test get in.
76       */
77      @Test
78      void testGetIn() {
79          final byte[] in = this.token.getIn();
80          Assertions.assertNotNull(in);
81          Assertions.assertArrayEquals(NegotiateTokenTest.IN_BYTES, in);
82      }
83  
84      /**
85       * Test get in returns clone.
86       */
87      @Test
88      void testGetInReturnsClone() {
89          final byte[] in1 = this.token.getIn();
90          final byte[] in2 = this.token.getIn();
91          Assertions.assertNotSame(in1, in2);
92      }
93  
94      /**
95       * Test set out.
96       */
97      @Test
98      void testSetOut() {
99          final byte[] newOut = { 0x07, 0x08 };
100         this.token.setOut(newOut);
101         Assertions.assertArrayEquals(newOut, this.token.getOut());
102     }
103 
104     /**
105      * Test set out null.
106      */
107     @Test
108     void testSetOutNull() {
109         this.token.setOut(null);
110         Assertions.assertNull(this.token.getOut());
111     }
112 
113     /**
114      * Test set and get subject.
115      */
116     @Test
117     void testSetAndGetSubject() {
118         final Subject subject = new Subject();
119         this.token.setSubject(subject);
120         Assertions.assertEquals(subject, this.token.getSubject());
121         Assertions.assertEquals(subject, this.token.getCredentials());
122     }
123 
124     /**
125      * Test set and get principal.
126      */
127     @Test
128     void testSetAndGetPrincipal() {
129         final Principal principal = () -> "testprincipal";
130         this.token.setPrincipal(principal);
131         Assertions.assertEquals(principal, this.token.getPrincipal());
132     }
133 
134     /**
135      * Test is remember me false.
136      */
137     @Test
138     void testIsRememberMeFalse() {
139         Assertions.assertFalse(this.token.isRememberMe());
140     }
141 
142     /**
143      * Test is remember me true.
144      */
145     @Test
146     void testIsRememberMeTrue() {
147         final NegotiateToken rememberToken = new NegotiateToken(NegotiateTokenTest.IN_BYTES,
148                 NegotiateTokenTest.OUT_BYTES, "conn-001", "Negotiate", false, true, null);
149         Assertions.assertTrue(rememberToken.isRememberMe());
150     }
151 
152     /**
153      * Test get host.
154      */
155     @Test
156     void testGetHost() {
157         Assertions.assertEquals("localhost", this.token.getHost());
158     }
159 
160     /**
161      * Test get host null.
162      */
163     @Test
164     void testGetHostNull() {
165         final NegotiateToken nullHostToken = new NegotiateToken(NegotiateTokenTest.IN_BYTES,
166                 NegotiateTokenTest.OUT_BYTES, "conn-001", "Negotiate", false, false, null);
167         Assertions.assertNull(nullHostToken.getHost());
168     }
169 
170     /**
171      * Test create info.
172      */
173     @Test
174     void testCreateInfo() {
175         final java.util.Set<java.security.Principal> principals = new java.util.HashSet<>();
176         principals.add(() -> "testUser");
177         final Subject subject = new Subject(false, principals, new java.util.HashSet<>(), new java.util.HashSet<>());
178         this.token.setSubject(subject);
179         final org.apache.shiro.authc.AuthenticationInfo info = this.token.createInfo();
180         Assertions.assertNotNull(info);
181         Assertions.assertNotNull(info.getCredentials());
182         Assertions.assertNotNull(info.getPrincipals());
183     }
184 }
