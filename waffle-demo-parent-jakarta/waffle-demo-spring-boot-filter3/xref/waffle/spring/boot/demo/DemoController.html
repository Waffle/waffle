7   package waffle.spring.boot.demo;
8   
9   import java.util.stream.Collectors;
10  
11  import org.springframework.security.core.Authentication;
12  import org.springframework.security.core.GrantedAuthority;
13  import org.springframework.web.bind.annotation.GetMapping;
14  import org.springframework.web.bind.annotation.RestController;
15  
16  /**
17   * Spring Boot Demo Rest Controller.
18   */
19  @RestController
20  public class DemoController {
21  
22      /**
23       * Spring Boot Demo Get Mapper.
24       *
25       * @param auth
26       *            security authentication object
27       *
28       * @return response text
29       */
30      @GetMapping
31      public String demo(final Authentication auth) {
32          return "Hello, %s. You have authorities: %s".formatted(auth.getPrincipal(),
33                  auth.getAuthorities().stream().map(GrantedAuthority::getAuthority).collect(Collectors.joining(", ")));
34      }
35  
36  }
